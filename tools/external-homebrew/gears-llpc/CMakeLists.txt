cmake_minimum_required(VERSION 3.21)
project(ToucheGearsCompiler LANGUAGES C CXX)
# Installed MSVC has only Russian resources. CMake misdetects its UTF-8 prefix;
# specify the observed prefix so Ninja records actual header dependencies.
set(CMAKE_CL_SHOWINCLUDES_PREFIX "Примечание: включение файла: ")
if(NOT GEARS_SOURCES)
  message(FATAL_ERROR "GEARS_SOURCES must name the pinned isolated source directory")
endif()
set(ICD_BUILD_LLPC ON)
set(LLPC_BUILD_TOOLS ON)
set(LLPC_BUILD_TESTS OFF)
set(LLPC_BUILD_LLVM_TOOLS ON)
set(LLPC_IS_STANDALONE ON)
set(LLPC_RAY_TRACING OFF CACHE BOOL "No ray tracing required by Gears")
set(PAL_SOURCE_DIR "${GEARS_SOURCES}/pal")
set(LLPC_LLVM_SRC_PATH "${GEARS_SOURCES}/llvm-project/llvm")
set(SPIRV_HEADERS_PATH "${GEARS_SOURCES}/third_party/SPIRV-tools/external/SPIRV-Headers")
set(LLPC_CLIENT_INTERFACE_MAJOR_VERSION 999999)
set(PAL_CLIENT_INTERFACE_MAJOR_VERSION 999999)
set(GPURT_CLIENT_INTERFACE_MAJOR_VERSION 999999)
set(TARGET_ARCHITECTURE_ENDIANESS LITTLE)
set(LLVM_TARGETS_TO_BUILD "AMDGPU;X86" CACHE STRING "Bounded target set")
set(LLVM_INCLUDE_TESTS OFF CACHE BOOL "No compiler tests in initial build")
set(LLVM_ENABLE_ASSERTIONS ON CACHE BOOL "Check compiler contracts")
set(LLVM_PARALLEL_COMPILE_JOBS 2 CACHE STRING "Bound memory on 16 GiB host")
set(LLVM_PARALLEL_LINK_JOBS 1 CACHE STRING "One link at a time")
set(LLVM_ENABLE_ZLIB OFF CACHE BOOL "No optional compression")
set(LLVM_ENABLE_ZSTD OFF CACHE BOOL "No optional compression")
set(LLVM_ENABLE_LIBXML2 OFF CACHE BOOL "No optional XML dependency")
set(LLVM_ENABLE_DIA_SDK OFF CACHE BOOL "No Windows PDB/DIA support or ATL needed for PAL shader compilation" FORCE)
add_library(khronos_vulkan_interface INTERFACE)
target_include_directories(khronos_vulkan_interface INTERFACE "${GEARS_SOURCES}/Vulkan-Headers/include")
target_compile_definitions(khronos_vulkan_interface INTERFACE EXTERNAL_VULKAN_HEADERS=1)
add_subdirectory("${GEARS_SOURCES}/third_party/metrohash" metrohash)
add_subdirectory("${GEARS_SOURCES}/third_party/cwpack" cwpack)
add_subdirectory("${GEARS_SOURCES}/llpc" llpc)
